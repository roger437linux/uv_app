import requests

url = 'https://servicodados.ibge.gov.br/api/v1/localidades/municipios'
response = requests.get(f"{url}", timeout=30)

# Verificar se a requisição HTTP retornou um erro e gerar exceção automaticamente
response.raise_for_status()

municipios = response.json()
print(municipios)



#------------------------------------------------------------------
#                       TRANSFORMAÇÃO
#------------------------------------------------------------------

✅ Transforma cada município em um registro tabular

dados = []

❌ Trata hierarquias que podem estar ausentes

for municipio in municipios:    
    microrregiao = municipio.get("microrregiao") or {}
    mesorregiao = microrregiao.get("mesorregiao") or {}
    uf = mesorregiao.get("UF") or {}
    regiao = uf.get("regiao") or {}

    dados.append({
        "id": municipio.get("id"),
        "nome_municipio": municipio.get("nome"),
        "nome_microrregiao": microrregiao.get("nome"),
        "nome_mesorregiao": mesorregiao.get("nome"),
        "uf_nome": uf.get("nome"),
        "uf_sigla": uf.get("sigla"),
        "regiao_nome": regiao.get("nome"),
        "regiao_sigla": regiao.get("sigla")
    })



#------------------------------------------------------------------
#                       JSON em DATAFRAME
#------------------------------------------------------------------


import pandas as pd

df = pd.DataFrame(
    dados,
    columns=[
        "id",
        "nome_municipio",
        "nome_microrregiao",
        "nome_mesorregiao",
        "uf_nome",
        "uf_sigla",
        "regiao_nome",
        "regiao_sigla"
    ]
)
